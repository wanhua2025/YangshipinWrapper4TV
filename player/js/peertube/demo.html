📝 使用示例 (ES Module)

// 引入 CSS // <link rel="stylesheet" href="./peertube/peertube-player.css"> import PeerTubeStandalonePlayer from './peertube/peertube-standalone.js'; const player = new PeerTubeStandalonePlayer('#playerContainer', { autoplay: false, controls: true, muted: false, loop: false, p2pEnabled: true, theme: 'lucide', // 事件回调 ready: () => console.log('播放器就绪'), play: () => console.log('开始播放'), pause: () => console.log('暂停'), ended: () => console.log('播放结束'), timeupdate: () => updateTimeDisplay(), error: (err) => console.error('错误:', err) }); // 加载 HLS 视频 await player.load('http://example.com/video/index.m3u8', { title: '我的视频', resolution: 720 }); // 控制方法 player.play(); player.pause(); player.currentTime(30); // 跳转到30秒 player.playbackRate(1.5); // 1.5倍速 player.volume(0.8); // 设置音量 0-1 player.requestFullscreen(); // 全屏 player.requestPictureInPicture(); // 画中画 player.showStats(); // 显示P2P统计 player.destroy(); // 销毁播放器
